마지막 동기화 기록 없음

연동 원리 및 Apps Script 조회(doGet) 설정 가이드

1. 자동 기록: 갤럭시 MacroDroid가 카드 결제 알림을 감지하면 Google Apps Script(doPost)로 보내 구글 시트에 누적합니다.

2. 가계부 동기화: 이 앱의 가져오기 버튼을 누르면 ?key=... 주소로 시트 데이터를 조회해 중복 없이 자동으로 합쳐 줍니다.

3. Apps Script에 추가할 조회 코드: 기존 스크립트 맨 아래에 아래 doGet 함수를 추가하고 배포 관리 → 새 버전으로 배포하세요. 

function doGet(e) {
  if (!e || e.parameter.key !== CONFIG.SECRET_KEY) {
    return jsonResponse({ status: 'error', message: 'Invalid key' });
  }
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.SHEET_LOG);
  if (!sheet || sheet.getLastRow() < 2) return jsonResponse({ status: 'ok', rows: [] });
  const vals = sheet.getRange(2, 1, sheet.getLastRow() - 1, 7).getValues();
  const rows = vals.filter(r => r[0]).map(r => ({
    date: String(r[0]), time: String(r[1] || '00:00'),
    card: String(r[2] || '기타카드'), merchant: String(r[3] || ''),
    amount: Number(r[4]) || 0, type: String(r[5] || '지출'), raw: String(r[6] || '')
  }));
  return jsonResponse({ status: 'ok', rows });
}
